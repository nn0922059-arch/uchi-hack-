function wildcard(str, rule) {
    var escapeRegex = (str) => str.replace(/([.*+?^=!:${}()|\[\]\/\\])/g, "\\$1");
    return new RegExp("^" + rule.split("*").map(escapeRegex).join(".*") + "$").test(str);
}

        }

            from { text-shadow: 0 0 4px #f0f, 0 0 10px #f0f, 0 0 20px #f0f; }
            to   { text-shadow: 0 0 6px #f0f, 0 0 16px #f0f, 0 0 32px #0ff, 0 0 64px #0ff; }

    document.head.appendChild(style);

    /* ================= БЕЗОПАСНАЯ ПЕРЕКРАСКА UI ================= */
    function applyNeonTheme() {
        if (!document.getElementById('neon-custom-overrides')) {
            const overrideStyles = document.createElement('style');
            overrideStyles.id = 'neon-custom-overrides';
            overrideStyles.textContent = `
                .neon-status-bar {
                    background: rgba(10, 0, 20, 0.85) !important;
                    border: 1px solid #0ff !important;
                    box-shadow: 0 0 15px rgba(0, 255, 255, 0.6), inset 0 0 10px rgba(0, 255, 255, 0.3) !important;
                    border-radius: 20px !important;
                    padding: 8px 20px !important;
                    backdrop-filter: blur(4px) !important;
                    font-family: 'Orbitron', sans-serif !important;
                }
                .neon-status-bar * {
                    color: #0ff !important;
                    text-shadow: 0 0 5px #0ff, 0 0 10px #0ff !important;
                }
                .neon-btn {
                    background: rgba(255, 0, 255, 0.1) !important;
                    border: 1px solid #f0f !important;
                    box-shadow: 0 0 12px rgba(255, 0, 255, 0.6), inset 0 0 8px rgba(255, 0, 255, 0.2) !important;
                    border-radius: 20px !important;
                    color: #fff !important;
                    text-shadow: 0 0 8px #f0f !important;
                    font-family: 'Orbitron', sans-serif !important;
                    cursor: pointer !important;
                    /* Убрали transition, чтобы кнопка не вызывала событий при перерисовке */
                }
                .neon-btn * {
                    color: #fff !important;
                    text-shadow: 0 0 8px #f0f !important;
                }
                .neon-btn:hover {
                    background: rgba(255, 0, 255, 0.3) !important;
                    box-shadow: 0 0 20px rgba(255, 0, 255, 0.9), inset 0 0 15px rgba(255, 0, 255, 0.4) !important;
                }
            `;
            document.head.appendChild(overrideStyles);
        }

        document.querySelectorAll('body *').forEach(el => {
            if (el.classList.contains('neon-status-bar') || el.classList.contains('neon-btn')) return;

            // Плашка по центру
            if (el.children.length === 0 && el.textContent.includes('UchiHack')) {
                let bar = el.parentElement;
                for(let i=0; i<3; i++) {
                    if(!bar || bar === document.body) break;
                    if (bar.textContent.includes('UchiHack') && bar.textContent.includes('Статус')) {
                        bar.classList.add('neon-status-bar');
                        break;
                    }
                    bar = bar.parentElement;
                }
            }
            
            // Кнопки
            const text = el.textContent.trim();
            if (el.children.length === 0 && (text === 'Решить карточку' || text === 'Решить задание')) {
                let btn = el.closest('div, button, a');
                if (btn && btn !== document.body) {
                    btn.classList.add('neon-btn');
                }
            }
        });
    }
    /* ====================================================== */

    // Создаем экран загрузки
    const ui = document.createElement('div');
    ui.id = 'neon-injector-ui';
    ui.innerHTML = `
        <div class="neon-title">UCHIHA</div>
        <div class="neon-subtitle">INJECTOR // LOADING</div>
        <div class="neon-custom-tag">tec53992newban</div>
        <div class="neon-bar"><div class="neon-bar-fill" id="neon-fill"></div></div>
        <div class="neon-status" id="neon-status">Инициализация...</div>
    `;
    document.body.appendChild(ui);

    const fill = ui.querySelector('#neon-fill');
    const statusText = ui.querySelector('#neon-status');

    const setProgress = (p, text) => {
        fill.style.width = p + '%';
        if (text !== undefined) statusText.textContent = text;
    };
    
    const destroyUI = () => {
        ui.classList.add('neon-hidden');
        setTimeout(() => { ui.remove(); }, 600);
    };

    var root = "https://raw.githubusercontent.com/theairblow/hackpack/main/uchihack/";

    setProgress(5, 'Загрузка конфигурации...');
    var config;
    try {
        const rawConfig = await fetch(root + "injector/config.json");
        config = await rawConfig.json();
    } catch (e) {
        setProgress(100, 'Ошибка конфига: ' + e.message);
        setTimeout(destroyUI, 2000);
        return;
    }

    setProgress(15, 'Проверка соответствия URL...');
    var glFound = false;
    var js = [];
    for (const item of config.injector) {
        var found = false;
        for (const url of item.urls) {
            if (wildcard(location.href, url)) { found = true; break; }
        }
        if (found) { js = item.scripts; glFound = true; break; }
    }

    if (!glFound) {
        setProgress(100, 'Скрипты для этого сайта не найдены');
        setTimeout(destroyUI, 1500);
        return;
    }

    setProgress(20, `Найдено скриптов: ${js.length}`);

    let i = 0;
    for (const item of js) {
        i++;
        setProgress(20 + (i / js.length) * 75, `Загрузка ${i}/${js.length}: ${item}`);
        try {
            const resp = await fetch(root + item);
            const raw = await resp.text();
            eval(raw);
        } catch (e) {
            console.error('Ошибка выполнения ' + item, e);
            statusText.textContent = 'Ошибка: ' + item;
        }
    }

    setProgress(100, 'Применение неоновой темы...');
    await new Promise(r => setTimeout(r, 800));
    
    /* ================= ЗАЩИТА ОТ АВТОКЛИКА ================= */
    // Временно блокируем все клики на 300 мс, пока применяются стили
    const preventClicks = (e) => {
        e.stopPropagation();
        e.preventDefault();
    };

    document.addEventListener('click', preventClicks, true);
    document.addEventListener('mousedown', preventClicks, true);
    document.addEventListener('mouseup', preventClicks, true);

    // Применяем стили
    applyNeonTheme();
    
    // Снимаем блокировку через 300 мс
    setTimeout(() => {
        document.removeEventListener('click', preventClicks, true);
        document.removeEventListener('mousedown', preventClicks, true);
        document.removeEventListener('mouseup', preventClicks, true);
    }, 300);
    /* ======================================================= */

    // На всякий случай применяем еще раз через секунду (если DOM обновился), но уже без блокировки
    setTimeout(applyNeonTheme, 1000);

    await new Promise(r => setTimeout(r, 400));
    destroyUI();
})();
